"""按用户决定更新16组固定配方及报告数据；不训练、不加载模型。"""
import hashlib
import json
import shutil
import sys
from pathlib import Path

ROOT = Path.cwd()
sys.path.insert(0, str(ROOT))
import numpy as np
import pandas as pd
from evaluation_functions import score_files
from experiments.deberta_lexical_mixed_pair.analyze_fusion import metrics

OUTPUT = ROOT / 'experiments/model_ensemble/outputs/run01_all_existing'
COMPARISON = OUTPUT / 'selection_comparison'
BACKUP = COMPARISON / 'original_policy_snapshot'
BACKUP.mkdir(parents=True, exist_ok=True)
for name in ['selected_recipes.json', 'recommended_combinations.csv', 'RESULTS.md', 'member_details.json',
             'S12_predictions.csv', 'S12_changes_vs_0938.csv']:
    if not (BACKUP / name).exists():
        shutil.copy2(OUTPUT / name, BACKUP / name)
if not (BACKUP / 'official_checks_S12').exists():
    shutil.copytree(OUTPUT / 'official_checks/S12', BACKUP / 'official_checks_S12')

recipes = json.loads((BACKUP / 'selected_recipes.json').read_text(encoding='utf-8'))
details = json.loads((BACKUP / 'member_details.json').read_text(encoding='utf-8'))
selected = pd.read_csv(BACKUP / 'recommended_combinations.csv', keep_default_na=False).set_index('code')
comparison = pd.read_csv(COMPARISON / 'comparison.csv', keep_default_na=False).set_index('scheme')
choices = pd.read_csv(COMPARISON / 'member_selection.csv', keep_default_na=False).set_index('id')
dev = pd.read_csv(ROOT / 'data/dev/dico_nli_dev_track1_participant_labeled.csv', keep_default_na=False)
FULL = ['FORWARD_ENTAILMENT', 'BACKWARD_ENTAILMENT', 'EQUIVALENCE', 'NEGATIVE_OTHER']
LABELS = list('FBEN')
gold = dev.label.map(dict(zip(FULL, range(4)))).to_numpy()
pairs = np.array([group.index.to_numpy() for _, group in dev.groupby('pair_id', sort=False) if len(group) == 2])
base = pd.read_csv(details['M059']['prediction_file'], keep_default_na=False).set_index('instance_id').loc[dev.instance_id]
baseline = base.pair_prediction.map(dict(zip(LABELS, range(4)))).to_numpy()
accepted = []
kept = []


def predictions(path):
    frame = pd.read_csv(path, keep_default_na=False)
    assert frame.instance_id.is_unique and set(frame.instance_id) == set(dev.instance_id)
    return frame.set_index('instance_id').loc[dev.instance_id].reset_index()


for code, recipe in recipes.items():
    row = comparison.loc[code]
    accept = row.pair_aware_delta_pp >= -1e-10
    recipe['selection_policy'] = 'pair_aware' if accept else 'retain_original'
    recipe['selection_policy_basis'] = 'User accepted uniform pair-aware unless historical ensemble pair-aware F1 decreased; fixed members and fusion.'
    if accept:
        accepted.append(code)
        new_predictions = predictions(COMPARISON / code / 'all_pair_selection/predictions_with_scores.csv')
        records = []
        for method, column in [('independent', 'prediction'), ('pair_aware', 'pair_prediction')]:
            values = new_predictions[column].map(dict(zip(LABELS, range(4)))).to_numpy()
            records.append(dict(method=method, **metrics(gold, values, pairs)))
        recipe['same_scores_metrics'] = records
        for member in recipe['members']:
            choice = choices.loc[member['id']]
            member['original_selection'] = member.get('selection')
            member['original_selected_epoch'] = details[member['id']]['selected_epoch']
            member['selection'] = 'best Dev pair-aware weighted F1; earliest epoch on ties'
            member['selected_epoch'] = int(choice.all_pair_epoch)
            member['prediction_file'] = choice.all_pair_prediction
            member['prediction_sha256'] = hashlib.sha256(Path(member['prediction_file']).read_bytes()).hexdigest()
            if int(choice.old_epoch) != int(choice.all_pair_epoch):
                frame = predictions(member['prediction_file'])
                stats = {method: metrics(gold, frame[column].map(dict(zip(LABELS, range(4)))).to_numpy(), pairs)
                         for method, column in [('independent', 'prediction'), ('pair_aware', 'pair_prediction')]}
                member['independent_f1'] = stats['independent']['weighted_f1']
                member['pair_f1'] = stats['pair_aware']['weighted_f1']
                member['N_correct'] = stats['pair_aware']['N_correct']
                member['pair_FBE_correct'] = stats['pair_aware']['FBE_correct']
                member['historical_checkpoint_paths'] = member.get('checkpoint_candidates_existing', [])
                member['checkpoint_candidates_existing'] = []
                member['checkpoint_found'] = False
                member['checkpoint_note'] = 'Only historical epoch prediction CSV verified; corresponding epoch22 weights not confirmed saved. Retrain and select by pair-aware.'
        current = {record['method']: record for record in records}
        for field, value in {
            'independent_f1': current['independent']['weighted_f1'], 'pair_f1': current['pair_aware']['weighted_f1'],
            'ind_N': current['independent']['N_correct'], 'pair_N': current['pair_aware']['N_correct'],
            'ind_FBE': current['independent']['FBE_correct'], 'pair_FBE': current['pair_aware']['FBE_correct'],
            'ind_non_N_to_N': current['independent']['non_N_to_N'],
        }.items():
            selected.loc[code, field] = value
        after = new_predictions.pair_prediction.map(dict(zip(LABELS, range(4)))).to_numpy()
        selected.loc[code, 'fixed_vs_0938'] = int(((baseline != gold) & (after == gold)).sum())
        selected.loc[code, 'broken_vs_0938'] = int(((baseline == gold) & (after != gold)).sum())
        if code == 'S12':
            title = '几何平均：统一pair-aware选模（原双目标方案）'
            recipe['title'] = title
            selected.loc[code, 'title'] = title
            restored = pd.read_csv(BACKUP / 'S12_predictions.csv', keep_default_na=False)
            assert restored.instance_id.tolist() == new_predictions.instance_id.tolist()
            for column in ['logit_F', 'logit_B', 'logit_E', 'logit_N', 'prediction', 'pair_prediction']:
                restored[column] = new_predictions[column]
            restored.to_csv(OUTPUT / 'S12_predictions.csv', index=False, encoding='utf-8-sig')
            changes = restored.loc[baseline != after].copy()
            changes['baseline_prediction'] = np.array(LABELS)[baseline[baseline != after]]
            changes.to_csv(OUTPUT / 'S12_changes_vs_0938.csv', index=False, encoding='utf-8-sig')
            for method, column in [('independent', 'prediction'), ('pair_aware', 'pair_prediction')]:
                folder = OUTPUT / 'official_checks/S12' / method
                path = folder / 'predictions.csv'
                pd.DataFrame({'instance_id': dev.instance_id, 'label': restored[column].map(dict(zip(LABELS, FULL)))}).to_csv(path, index=False)
                score = score_files(ROOT / 'data/dev/dico_nli_dev_track1_reference.csv', path, output_dir=folder)
                assert abs(score.weighted_f1 - current[method]['weighted_f1']) < 1e-12
    else:
        kept.append(code)
        for member in recipe['members']:
            member['selected_epoch'] = details[member['id']]['selected_epoch']
    selected.loc[code, 'selection_policy'] = recipe['selection_policy']
    selected.loc[code, 'original_independent_f1'] = float(row.original_independent)
    selected.loc[code, 'original_pair_f1'] = float(row.original_pair_aware)
    selected.loc[code, 'all_checkpoints_found'] = all(m.get('checkpoint_found', False) for m in recipe['members'])

(OUTPUT / 'selected_recipes.json').write_text(json.dumps(recipes, ensure_ascii=False, indent=2), encoding='utf-8')
selected.to_csv(OUTPUT / 'recommended_combinations.csv', encoding='utf-8-sig')
(COMPARISON / 'accepted_policy.json').write_text(json.dumps({
    'uniform_pair_aware': accepted, 'retain_original': kept,
    'criterion': 'historical ensemble pair-aware F1 did not decrease',
    'no_training_executed': True, 'original_snapshots': str(BACKUP),
}, ensure_ascii=False, indent=2), encoding='utf-8')

for code in ['S01', 'S11', 'S12', 'S16']:
    path = ROOT / 'experiments/ensemble_selected' / (code + '.py')
    text = path.read_text(encoding='utf-8')
    text = text.replace("'select_by': 'independent'", "'select_by': 'pair_aware'").replace("'select_by': 'joint'", "'select_by': 'pair_aware'")
    text = text.replace("RUN_NAME = 'run01_retrain'", "RUN_NAME = 'run02_pair_select'")
    path.write_text(text, encoding='utf-8')
print('UNIFORM PAIR:', ', '.join(accepted))
print('KEEP ORIGINAL:', ', '.join(kept))
print('S12:', selected.loc['S12', ['independent_f1', 'pair_f1', 'pair_N', 'pair_FBE', 'fixed_vs_0938', 'broken_vs_0938']].to_dict())
